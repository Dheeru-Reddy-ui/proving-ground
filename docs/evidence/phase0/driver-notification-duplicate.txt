"""Repro: AltTester-Driver 2.3.2 NotificationHandler delivers a LOG notification twice
when a listener is added with overwrite=True (the AltDriver.add_notification_listener default)."""
from alttester._websocket import NotificationHandler
from alttester.commands.Notifications.notification_type import NotificationType

msg = {"commandName": "logNotification", "data": '{"message": "hi", "stack_trace": "", "level": 3}'}
for overwrite in (True, False):
    handler = NotificationHandler()
    got = []
    handler.add_notification_listener(NotificationType.LOG, got.append, overwrite=overwrite)
    handler.handle_notification(msg)
    print(f"overwrite={overwrite}: callbacks registered={len(handler._notification_callbacks[NotificationType.LOG])}, deliveries of ONE log={len(got)}")

## Output
overwrite=True: callbacks registered=2, deliveries of ONE log=2
overwrite=False: callbacks registered=1, deliveries of ONE log=1
